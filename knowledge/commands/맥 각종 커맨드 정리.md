# 맥 각종 커맨드 정리

## 선행 필요 조건

- honebrew install 하기
	+ https://salmonpack.tistory.com/52 참고
	+ https://engineeringfordatascience.com/posts/setting_up_a_macbook_for_data_science/ 참고 -> DS 기본 세팅

- 앱 강제 종료하는 방법
	- cmd + option + esc

- 파이썬 가상환경 관련
	- source .venv/bin/activate
	- deactivate

- aws config 관련 명령어
	+ cd ~/.aws
	+ touch credentials # 해당 디렉토리에 파일 생성
	+ cat credentials # 파일 내용 보기
	+ vim credentials # 파일 수정하기

- 터미널 관련 명령어
	+ chsh -s /bin/bash # bash로 돌아가기
	+ chsh -s /bin/zsh # zsh로 돌아가기
	+ subl ~/.zshrc # zsh 환경설정 수정
	+ 서브라임 깔려있다는 전제 하에
		+ subl .
			+ -> 해당 디렉토리 내에서 서브라임 오픈
			+ -> 디렉토리 구조 바로 확인 가능
	+ vscode 깔려있다는 전제 하에
		+ code .
			+ -> 해당 디렉토리 내에서 vscode 오픈
			+ -> 디렉토리 구조 바로 확인 가능

- 기타 명령어
	+ vim 상태에서 i -> 수정모드 / esc -> 컨트롤 모드
	+ 컨트롤 모드에서 :wq 하면 저장 및 파일 나가기
	+ ls = 현 디렉토리 내 존재하는 파일/폴더 모두 보기
	+ pwd = 현 디렉토리 풀 네임 명시 (복붙 가능)
	+ cd ~/ = 홈으로 무조건 컴백

- 크롬 명령어
	+ command + l -> 주소창으로 커서 이동, 바로 검색어 입력 가능
	+ option + command + 화살표 -> 탭 이동
	+ command + ` -> 동일 앱 이전 창으로 이동

# ~/.bash_profile 내용

``` batch

# Setting PATH for Python 3.12
# The original version is saved in .bash_profile.pysave
PATH="/Library/Frameworks/Python.framework/Versions/3.12/bin:${PATH}"
export PATH
eval "$(/opt/homebrew/bin/brew shellenv)"
export PATH="/opt/homebrew/opt/openjdk@11/bin:$PATH"
# >>> conda initialize >>>
# !! Contents within this block are managed by 'conda init' !!
__conda_setup="$('/Users/woojoo.jang/opt/miniconda/bin/conda' 'shell.bash' 'hook' 2> /dev/null)"
if [ $? -eq 0 ]; then
    eval "$__conda_setup"
else
    if [ -f "/Users/woojoo.jang/opt/miniconda/etc/profile.d/conda.sh" ]; then
        . "/Users/woojoo.jang/opt/miniconda/etc/profile.d/conda.sh"
    else
        export PATH="/Users/woojoo.jang/opt/miniconda/bin:$PATH"
    fi
fi
unset __conda_setup
# <<< conda initialize <<<


```


# ~/.zshrc 내용

``` batch
# Enable Powerlevel10k instant prompt. Should stay close to the top of ~/.zshrc.
# Initialization code that may require console input (password prompts, [y/n]
# confirmations, etc.) must go above this block; everything else may go below.
if [[ -r "${XDG_CACHE_HOME:-$HOME/.cache}/p10k-instant-prompt-${(%):-%n}.zsh" ]]; then
  source "${XDG_CACHE_HOME:-$HOME/.cache}/p10k-instant-prompt-${(%):-%n}.zsh"
fi

export PATH="/opt/homebrew/opt/openjdk@11/bin:$PATH"

if [ -f ~/.bash_profile ]; then
  . ~/.bash_profile
fi

source ~/powerlevel10k/powerlevel10k.zsh-theme

# To customize prompt, run `p10k configure` or edit ~/.p10k.zsh.
[[ ! -f ~/.p10k.zsh ]] || source ~/.p10k.zsh

code () { VSCODE_CWD="$PWD" open -n -b "com.microsoft.VSCode" --args $* ;}

```
